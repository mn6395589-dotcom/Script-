local function activateObject(obj)
    local char = Player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return false end
    
    local root = char.HumanoidRootPart
    
    local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt")
    if prompt then
        fireproximityprompt(prompt)
        return true
    end
    
    local cd = obj:FindFirstChildWhichIsA("ClickDetector")
    if cd then
        fireclickdetector(cd)
        return true
    end
    
    if obj:IsA("BasePart") then
        firetouchinterest(root, obj, 0)
        task.wait(0.05)
        firetouchinterest(root, obj, 1)
        return true
    end
    
    if obj:IsA("BasePart") then
        local oldPos = root.CFrame
        root.CFrame = CFrame.new(obj.Position + Vector3.new(0, 1, 0))
        task.wait(0.1)
        root.CFrame = oldPos
        return true
    end
    
    return false
end

local function fireEvent(eventName, ...)
    for _, v in pairs(ReplicatedStorage:GetDescendants()) do
        if v:IsA("RemoteEvent") and v.Name:lower():find(eventName:lower()) then
            v:FireServer(...)
            return true
        end
    end
    return false
end

-- AUTO FARM
-- ============================================
task.spawn(function()
    while task.wait(0.3) do
        if isVerified and farming then
            pcall(function()
                local char = Player.Character
                if not char then return end
                
                local root = char.HumanoidRootPart
                local nearest = nil
                local minDist = 50
                
                for _, obj in pairs(Workspace:GetDescendants()) do
                    local pos = nil
                    local target = nil
                    
                    if obj:IsA("ProximityPrompt") then
                        target = obj.Parent
                        pos = target:IsA("BasePart") and target.Position or target:GetPivot().Position
                    elseif obj:IsA("ClickDetector") then
                        target = obj.Parent
                        pos = target:IsA("BasePart") and target.Position or target:GetPivot().Position
                    elseif obj:IsA("Model") and obj.PrimaryPart and obj.PrimaryPart.Size.Magnitude < 30 then
                        target = obj
                        pos = obj.PrimaryPart.Position
                    elseif obj:IsA("BasePart") and obj.Size.Magnitude < 20 and obj.CanCollide and obj.Transparency < 0.9 then
                        target = obj
                        pos = obj.Position
                    end
                    
                    if pos and target then
                        local dist = (root.Position - pos).Magnitude
                        if dist < minDist then
                            minDist = dist
                            nearest = target
                        end
                    end
                end
                
                if nearest then
                    local pos = nearest:IsA("BasePart") and nearest.Position or nearest:GetPivot().Position
                    
                    if (root.Position - pos).Magnitude > 5 then
                        root.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
                    end
                    
                    task.wait(0.1)
                    activateObject(nearest)
                    fireEvent("Farm")
                    fireEvent("Harvest")
                    fireEvent("Collect")
                    fireEvent("Plant")
                    fireEvent("Click")
                end
            end)
        end
    end
end)

                            print("🥷 Steal: " .. obj.Name)

                        end
                    end
                end
                
                for _, player in pairs(Players:GetPlayers()) do
                    if player ~= Player then
                        local backpack = player:FindFirstChild("Backpack")
                        if backpack then
                            for _, tool in pairs(backpack:GetChildren()) do
                                if tool:IsA("Tool") then
                                    tool.Parent = Player.Backpack
                                    print("🥷 Steal tu " .. player.Name)
                                    return
                                end
                            end
                        end
                        
                        if player.Character then
                            for _, tool in pairs(player.Character:GetChildren()) do
                                if tool:IsA("Tool") and tool.Name ~= "HumanoidRootPart" then
                                    tool.Parent = Player.Backpack
                                    print("🥷 Steal tool " .. player.Name)
                                    return
                                end
                            end
                        end
                    end
                end
            end)
        end
    end
end)
