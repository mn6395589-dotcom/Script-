local KeySystem = {}
KeySystem.ValidKey = "DEV_MINH 2005 DZ" -- Key mặc định
KeySystem.KeyChecked = false

-- Hàm kiểm tra key
function KeySystem:CheckKey(inputKey)
    if inputKey == self.ValidKey then
        self.KeyChecked = true
        return true
    else
        return false
    end
end

-- ============================================
-- MAIN SCRIPT
-- ============================================
function loadMainScript()
    local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()
    
    local Window = Rayfield:CreateWindow({
       Name = "⚔️ DEV_MINH 2005",
       LoadingTitle = "DEV_MINH 2005 Hub",
       LoadingSubtitle = "Ktanana Auto Kill",
       ConfigurationSaving = {
          Enabled = false
       },
       KeySystem = false
    })
    
    -- ============================================
    -- TAB: AUTO KILL
    -- ============================================
    local CombatTab = Window:CreateTab("⚔️ Auto Kill", "crosshair")
    
    -- Hàm tìm và cầm kiếm
    local function equipSword()
       local plr = game.Players.LocalPlayer
       local char = plr.Character
       if not char then return nil end
       
       local backpack = plr:FindFirstChild("Backpack")
       local tool = char:FindFirstChildOfClass("Tool")
       
       if tool then return tool end
       
       if backpack then
          for _, item in pairs(backpack:GetChildren()) do
             if item:IsA("Tool") then
                item.Parent = char
                task.wait(0.2)
                return item
             end
          end
       end
       
       for _, obj in pairs(workspace:GetChildren()) do
          if obj:IsA("Tool") and obj:FindFirstChild("Handle") then
             local hrp = char:FindFirstChild("HumanoidRootPart")
             if hrp then
                local dist = (obj.Handle.Position - hrp.Position).Magnitude
                if dist < 20 then
                   hrp.CFrame = obj.Handle.CFrame
                   task.wait(0.3)
                   firetouchinterest(char, obj, 0)
                   task.wait(0.2)
                   return obj
                end
             end
          end
       end
       
       return nil
    end
    
    -- Hàm tìm địch
    local function getNearestEnemy(range)
       local plr = game.Players.LocalPlayer
       local char = plr.Character
       if not char then return nil end
       
       local nearest = nil
       local nearDist = range or 50
       local hrp = char:FindFirstChild("HumanoidRootPart")
       if not hrp then return nil end
       
       for _, target in pairs(game.Players:GetPlayers()) do
          if target ~= plr and target.Character then
             local tHum = target.Character:FindFirstChild("Humanoid")
             local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
             if tHum and tHum.Health > 0 and tRoot then
                local dist = (tRoot.Position - hrp.Position).Magnitude
                if dist < nearDist then
                   nearDist = dist
                   nearest = target
                end
             end
          end
       end
       
       return nearest
    end
    
    -- Hàm tấn công
    local function attackTarget(target)
       local plr = game.Players.LocalPlayer
       local char = plr.Character
       if not char or not target or not target.Character then return end
       
       local tool = equipSword()
       if not tool then return end
       
       local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
       local mRoot = char:FindFirstChild("HumanoidRootPart")
       if not tRoot or not mRoot then return end
       
       mRoot.CFrame = CFrame.new(mRoot.Position, Vector3.new(tRoot.Position.X, mRoot.Position.Y, tRoot.Position.Z))
       
       pcall(function()
          tool:Activate()
       end)
       
       task.spawn(function()
          pcall(function()
             local replicated = game:GetService("ReplicatedStorage")
             
             local remotes = {
                replicated:FindFirstChild("Remotes"),
                replicated:FindFirstChild("RemoteEvent"),
                replicated:FindFirstChild("RemoteFunction"),
                replicated:FindFirstChild("Events"),
                replicated:FindFirstChild("DamageEvent"),
                replicated:FindFirstChild("HitEvent"),
                replicated:FindFirstChild("AttackEvent"),
                replicated:FindFirstChild("SwingEvent"),
                replicated:FindFirstChild("CombatEvent")
             }
             
             for _, remote in pairs(remotes) do
                if remote then
                   if remote:IsA("RemoteEvent") then
                      remote:FireServer(target.Character)
                   elseif remote:IsA("RemoteFunction") then
                      remote:InvokeServer(target.Character)
                   end
                   
                   if remote:IsA("RemoteEvent") then
                      remote:FireServer(tool, target.Character)
                   elseif remote:IsA("RemoteFunction") then
                      remote:InvokeServer(tool, target.Character)
                   end
                end
             end
             
             if tool:FindFirstChild("RemoteEvent") then
                tool.RemoteEvent:FireServer(target.Character)
             end
             if tool:FindFirstChild("Activate") then
                tool.Activate:FireServer()
             end
          end)
       end)
    end
    
    -- Auto Kill
    local autoKill = false
    local killRange = 30
    
    CombatTab:CreateToggle({
       Name = "⚔️ AUTO KILL",
       CurrentValue = false,
       Flag = "autokill",
       Callback = function(Value)
          autoKill = Value
          if Value then
             task.spawn(function()
                while autoKill do
                   task.wait()
                   
                   local enemy = getNearestEnemy(killRange)
                   if enemy then
                      local plr = game.Players.LocalPlayer
                      local char = plr.Character
                      if char then
                         local mRoot = char:FindFirstChild("HumanoidRootPart")
                         local tRoot = enemy.Character:FindFirstChild("HumanoidRootPart")
                         if mRoot and tRoot then
                            mRoot.CFrame = tRoot.CFrame + Vector3.new(0, 2, -3)
                            task.wait(0.1)
                            
                            for i = 1, 5 do
                               if not autoKill then break end
                               attackTarget(enemy)
                               task.wait(0.2)
                            end
                         end
                      end
                   end
                   
                   task.wait(0.5)
                end
             end)
          end
       end
    })
    
    CombatTab:CreateSlider({
       Name = "📏 Phạm vi",
       Range = {10, 100},
       Increment = 5,
       Suffix = "studs",
       CurrentValue = 30,
       Flag = "range",
       Callback = function(Value)
          killRange = Value
       end
    })
    
    CombatTab:CreateButton({
       Name = "🗡 Auto Cầm Kiếm",
       Callback = function()
          local tool = equipSword()
          if tool then
             Rayfield:Notify({
                Title = "✅ DEV_MINH 2005",
                Content = "Đã cầm: " .. tool.Name,
                Duration = 3,
                Image = "sword"
             })
          end
       end
    })
    
    -- ============================================
    -- TAB: PLAYER
    -- ============================================
    local PlayerTab = Window:CreateTab("👤 Nhân Vật", "user")
    
    PlayerTab:CreateSlider({
       Name = "🏃 Tốc Độ",
       Range = {16, 200},
       Increment = 1,
       Suffix = "Speed",
       CurrentValue = 16,
       Flag = "speed",
       Callback = function(Value)
          local char = game.Players.LocalPlayer.Character
          if char then
             local hum = char:FindFirstChild("Humanoid")
             if hum then hum.WalkSpeed = Value end
          end
       end
    })
    
    PlayerTab:CreateSlider({
       Name = "🦘 Sức Nhảy",
       Range = {50, 500},
       Increment = 1,
       Suffix = "Jump",
       CurrentValue = 50,
       Flag = "jump",
       Callback = function(Value)
          local char = game.Players.LocalPlayer.Character
          if char then
             local hum = char:FindFirstChild("Humanoid")
             if hum then hum.JumpPower = Value end
          end
       end
    })
    
    -- Fly
    local flying = false
    PlayerTab:CreateToggle({
       Name = "✈ Bay (Fly)",
       CurrentValue = false,
       Flag = "fly",
       Callback = function(Value)
          flying = Value
          local plr = game.Players.LocalPlayer
          local char = plr.Character or plr.CharacterAdded:Wait()
          local hum = char:WaitForChild("Humanoid")
          local hrp = char:WaitForChild("HumanoidRootPart")
          
          if Value then
             local bg = Instance.new("BodyGyro")
             bg.P = 9e4
             bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
             bg.CFrame = hrp.CFrame
             bg.Parent = hrp
             
             local bv = Instance.new("BodyVelocity")
             bv.Velocity = Vector3.new(0, 0, 0)
             bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
             bv.Parent = hrp
             
             hum.PlatformStand = true
             
             task.spawn(function()
                while flying do
                   task.wait()
                   bg.CFrame = workspace.CurrentCamera.CFrame
                   local moveDir = Vector3.new()
                   local flySpeed = 50
                   local uis = game:GetService("UserInputService")
                   
                   if uis:IsKeyDown(Enum.KeyCode.W) then moveDir += workspace.CurrentCamera.CFrame.LookVector end
                   if uis:IsKeyDown(Enum.KeyCode.S) then moveDir -= workspace.CurrentCamera.CFrame.LookVector end
                   if uis:IsKeyDown(Enum.KeyCode.A) then moveDir -= workspace.CurrentCamera.CFrame.RightVector end
                   if uis:IsKeyDown(Enum.KeyCode.D) then moveDir += workspace.CurrentCamera.CFrame.RightVector end
                   if uis:IsKeyDown(Enum.KeyCode.Space) then moveDir += Vector3.new(0, 1, 0) end
                   if uis:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir -= Vector3.new(0, 1, 0) end
                   
                   bv.Velocity = moveDir * flySpeed
                end
                bg:Destroy()
                bv:Destroy()
                hum.PlatformStand = false
             end)
          end
       end
    })
    
    -- God Mode
    PlayerTab:CreateToggle({
       Name = "🛡 God Mode",
       CurrentValue = false,
       Flag = "godmode",
       Callback = function(Value)
          if Value then
             task.spawn(function()
                while Value do
                   task.wait(0.1)
                   local char = game.Players.LocalPlayer.Character
                   if char then
                      local hum = char:FindFirstChild("Humanoid")
                      if hum and hum.Health < hum.MaxHealth then
                         hum.Health = hum.MaxHealth
                      end
                   end
                end
             end)
          end
       end
    })
    
    -- ============================================
    -- TAB: ESP
    -- ============================================
    local ESPTab = Window:CreateTab("👁 ESP", "eye")
    
    local espList = {}
    ESPTab:CreateToggle({
       Name = "👁 Bật ESP",
       CurrentValue = false,
       Flag = "esp",
       Callback = function(Value)
          local plr = game.Players.LocalPlayer
          if Value then
             for _, p in pairs(game.Players:GetPlayers()) do
                if p ~= plr and p.Character then
                   local hl = Instance.new("Highlight")
                   hl.FillColor = Color3.fromRGB(255, 0, 0)
                   hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                   hl.FillTransparency = 0.5
                   hl.Parent = p.Character
                   table.insert(espList, {player = p, highlight = hl})
                end
             end
             
             game.Players.PlayerAdded:Connect(function(p)
                p.CharacterAdded:Connect(function(char)
                   task.wait(0.5)
                   local hl = Instance.new("Highlight")
                   hl.FillColor = Color3.fromRGB(255, 0, 0)
                   hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                   hl.FillTransparency = 0.5
                   hl.Parent = char
                   table.insert(espList, {player = p, highlight = hl})
                end)
             end)
             
             game.Players.PlayerRemoving:Connect(function(p)
                for i, obj in pairs(espList) do
                   if obj.player == p then
                      if obj.highlight then obj.highlight:Destroy() end
                      table.remove(espList, i)
                      break
                   end
                end
             end)
          else
             for _, obj in pairs(espList) do
                if obj.highlight then obj.highlight:Destroy() end
             end
             espList = {}
          end
       end
    })
    
    -- ============================================
    -- TAB: KHÁC
    -- ============================================
    local MiscTab = Window:CreateTab("🔧 Khác", "settings")
    
    MiscTab:CreateButton({
       Name = "🔧 Infinite Yield",
       Callback = function()
          loadstring(game:HttpGet("https://raw.githubusercontent.com/EdgeIY/infiniteyield/master/source"))()
       end
    })
    
    MiscTab:CreateButton({
       Name = "🔄 Rejoin",
       Callback = function()
          game:GetService("TeleportService"):Teleport(game.PlaceId, game.Players.LocalPlayer)
       end
    })
    
    -- ============================================
    -- THÔNG BÁO
    -- ============================================
    Rayfield:Notify({
       Title = "⚔️ DEV_MINH 2005",
       Content = "Hub đã load thành công!",
       Duration = 5,
       Image = "check"
    })
    
    print("✅ DEV_MINH 2005 HUB - Loaded!")
end
