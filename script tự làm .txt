function loadMainScript()
    local plr = game.Players.LocalPlayer
    local prem = KeySystem.IsPremium
    
    local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()
    local Window = Rayfield:CreateWindow({
        Name = prem and "👑 DEV_MINH 2005 PRIME" or "⚔️ DEV_MINH 2005 VIP",
        LoadingTitle = "DEV_MINH 2005", LoadingSubtitle = "Auto Move Kill All",
        ConfigurationSaving = {Enabled = false}, KeySystem = false
    })
    
    local autoKill, killRange, moveSpeed = true, prem and 200 or 100, prem and 300 or 150
    local flying, godModeValue, killAllMode = false, false, false
    local currentTarget = nil
    
    -- Particles Premium
    if prem then task.spawn(function() while true do task.wait(0.5) pcall(function()
        local c = plr.Character if c and c:FindFirstChild("HumanoidRootPart") then
        local p = Instance.new("ParticleEmitter", c.HumanoidRootPart)
        p.Texture = "rbxasset://textures/particles/sparkles_main.dds"
        p.Rate = 50 p.Lifetime = NumberRange.new(0.5,1) task.wait(0.3) p:Destroy() end end) end end) end
    
    -- Equip
    local function equip()
        local c = plr.Character if not c then return nil end
        local tool = c:FindFirstChildOfClass("Tool") if tool then return tool end
        local bp = plr:FindFirstChild("Backpack")
        if bp then for _,v in bp:GetChildren() do if v:IsA("Tool") then
            local h = c:FindFirstChild("Humanoid") if h then h:EquipTool(v) task.wait(0.2) return v end end end end
        local hrp = c:FindFirstChild("HumanoidRootPart")
        if hrp then for _,v in workspace:GetDescendants() do if v:IsA("Tool") and v:FindFirstChild("Handle") then
            if (v.Handle.Position-hrp.Position).Magnitude<50 then hrp.CFrame=v.Handle.CFrame+Vector3.new(0,3,0)
            task.wait(0.2) firetouchinterest(hrp,v.Handle,0) firetouchinterest(hrp,v.Handle,1)
            task.wait(0.3) if c:FindFirstChildOfClass("Tool") then return c:FindFirstChildOfClass("Tool") end end end end end
        return nil
    end
    
    -- Get all enemies sorted by distance
    local function getAllEnemies(r)
        local c = plr.Character if not c then return {} end
        local hrp = c:FindFirstChild("HumanoidRootPart") if not hrp then return {} end
        local list = {}
        for _,t in game.Players:GetPlayers() do if t~=plr and t.Character then
            local h = t.Character:FindFirstChild("Humanoid")
            local tr = t.Character:FindFirstChild("HumanoidRootPart")
            if h and h.Health>0 and tr then
                local dist = (tr.Position-hrp.Position).Magnitude
                if dist < (r or 500) then table.insert(list,{target=t,dist=dist,root=tr}) end
            end end end
        table.sort(list,function(a,b) return a.dist<b.dist end)
        return list
    end
    
    -- Move to target smoothly
    local function moveToTarget(root)
        local c = plr.Character if not c then return end
        local hrp = c:FindFirstChild("HumanoidRootPart") if not hrp or not root then return end
        local hum = c:FindFirstChild("Humanoid") if not hum then return end
        
        -- Set speed
        hum.WalkSpeed = moveSpeed
        
        -- Move towards target
        local dir = (root.Position - hrp.Position).unit
        local targetPos = root.Position - dir * 3
        
        -- Teleport closer if too far
        local dist = (hrp.Position - root.Position).Magnitude
        if dist > 50 then
            hrp.CFrame = CFrame.new(targetPos + Vector3.new(0,2,0))
        else
            -- Walk smoothly
            hum:MoveTo(targetPos)
        end
        
        -- Look at target
        hrp.CFrame = CFrame.lookAt(hrp.Position, Vector3.new(root.Position.X, hrp.Position.Y, root.Position.Z))
        return dist
    end
    
    -- Attack single target
    local function atk(tg)
        local c = plr.Character if not c or not tg or not tg.Character then return false end
        local tool = equip() if not tool then return false end
        local tr = tg.Character:FindFirstChild("HumanoidRootPart")
        local mr = c:FindFirstChild("HumanoidRootPart")
        if not tr or not mr then return false end
        
        -- AOE attack for premium
        if prem then for _,t in game.Players:GetPlayers() do if t~=plr and t.Character then
            local troot = t.Character:FindFirstChild("HumanoidRootPart")
            if troot and (troot.Position-mr.Position).Magnitude<20 then
                task.spawn(function() pcall(function() tool:Activate() end) end) end end end end
        
        -- Position behind target
        mr.CFrame = tr.CFrame * CFrame.new(0,2,-2)
        task.wait(0.03)
        pcall(function() tool:Activate() end)
        
        -- Fire remotes
        task.spawn(function() pcall(function()
            for _,r in game.ReplicatedStorage:GetDescendants() do
                if r:IsA("RemoteEvent") then r:FireServer(tg.Character) r:FireServer(tool)
                elseif r:IsA("RemoteFunction") then r:InvokeServer(tg.Character) r:InvokeServer(tool) end
            end end) end)
        return true
    end
    
    -- Kill All function (sequential)
    local function killAllSequential()
        task.spawn(function()
            local enemies = getAllEnemies(500)
            for _,enemy in pairs(enemies) do
                if not killAllMode then break end
                local tg = enemy.target
                if tg and tg.Character then
                    local tr = tg.Character:FindFirstChild("HumanoidRootPart")
                    if tr then
                        -- Move to target
                        moveToTarget(tr)
                        task.wait(0.2)
                        -- Attack multiple times
                        for i = 1, 8 do
                            if not killAllMode then break end
                            atk(tg)
                            task.wait(prem and 0.1 or 0.2)
                        end
                    end
                end
                task.wait(0.1)
            end
            -- Loop back if still enabled
            if killAllMode then
                task.wait(0.5)
                killAllSequential()
            end
        end)
    end
    
    -- Main Auto Kill Loop (keeps moving to nearest)
    task.spawn(function()
        task.wait(1)
        equip()
        while task.wait(prem and 0.1 or 0.2) do
            if not autoKill then continue end
            local c = plr.Character
            if not c then task.wait(1) continue end
            local hum = c:FindFirstChild("Humanoid")
            if hum and hum.Health <= 0 then task.wait(2) continue end
            
            -- Find nearest enemy
            local enemies = getAllEnemies(killRange)
            if #enemies > 0 then
                local nearest = enemies[1]
                local tr = nearest.root
                if tr then
                    -- Move to target
                    moveToTarget(tr)
                    task.wait(0.15)
                    -- Attack
                    atk(nearest.target)
                end
            end
        end
    end)
    
    plr.CharacterAdded:Connect(function()
        if autoKill then task.wait(1) equip() end
        if killAllMode then task.wait(1) killAllSequential() end
    end)
    
    -- GUI
    local CTab = Window:CreateTab("⚔️ Auto Kill", "crosshair")
    
    CTab:CreateToggle({
        Name = "⚡ AUTO KILL (Tự di chuyển)",
        CurrentValue = true, Flag = "ak",
        Callback = function(v) autoKill = v if v then equip() end end
    })
    
    CTab:CreateSlider({
        Name = "📏 Phạm Vi Quét",
        Range = {10, prem and 500 or 200}, Increment = 10,
        Suffix = " studs", CurrentValue = killRange, Flag = "range",
        Callback = function(v) killRange = v end
    })
    
    CTab:CreateSlider({
        Name = "🏃 Tốc Độ Di Chuyển",
        Range = {50, prem and 500 or 300}, Increment = 10,
        Suffix = " speed", CurrentValue = moveSpeed, Flag = "movespd",
        Callback = function(v) moveSpeed = v end
    })
    
    CTab:CreateButton({
        Name = "🗡 Cầm Kiếm Ngay",
        Callback = function() equip() end
    })
    
    -- Kill All Toggle
    CTab:CreateToggle({
        Name = "💀 KILL ALL (Tự động di chuyển giết tất cả)",
        CurrentValue = false, Flag = "killall",
        Callback = function(v)
            killAllMode = v
            if v then
                autoKill = false -- Turn off single target mode
                equip()
                task.wait(0.5)
                killAllSequential()
                Rayfield:Notify({Title="💀 KILL ALL",Content="Đang di chuyển giết tất cả!",Duration=3,Image="skull"})
            end
        end
    })
    
    local PTab = Window:CreateTab("👤 Nhân Vật", "user")
    PTab:CreateSlider({Name="🏃 Tốc Độ Mặc Định",Range={16,prem and 500 or 200},Increment=1,CurrentValue=16,Flag="spd",
        Callback=function(v) if plr.Character then plr.Character:FindFirstChild("Humanoid").WalkSpeed=v end end})
    PTab:CreateSlider({Name="🦘 Sức Nhảy",Range={50,prem and 1000 or 500},Increment=1,CurrentValue=50,Flag="jmp",
        Callback=function(v) if plr.Character then plr.Character:FindFirstChild("Humanoid").JumpPower=v end end})
    
    PTab:CreateToggle({Name="✈ Bay (Fly) "..(prem and"[SIÊU TỐC]"or""),CurrentValue=false,Flag="fly",
        Callback=function(v)
            flying=v local c=plr.Character or plr.CharacterAdded:Wait() local h=c:WaitForChild("Humanoid") local hr=c:WaitForChild("HumanoidRootPart")
            if v then local bg=Instance.new("BodyGyro",hr) bg.P=9e4 bg.MaxTorque=Vector3.new(9e9,9e9,9e9) bg.CFrame=hr.CFrame
            local bv=Instance.new("BodyVelocity",hr) bv.MaxForce=Vector3.new(9e9,9e9,9e9) h.PlatformStand=true
            task.spawn(function() while flying do task.wait() bg.CFrame=workspace.CurrentCamera.CFrame local m=Vector3.new() local u=game.UserInputService
            local speed=prem and 150 or 50
            if u:IsKeyDown(Enum.KeyCode.W) then m+=workspace.CurrentCamera.CFrame.LookVector end
            if u:IsKeyDown(Enum.KeyCode.S) then m-=workspace.CurrentCamera.CFrame.LookVector end
            if u:IsKeyDown(Enum.KeyCode.A) then m-=workspace.CurrentCamera.CFrame.RightVector end
            if u:IsKeyDown(Enum.KeyCode.D) then m+=workspace.CurrentCamera.CFrame.RightVector end
            if u:IsKeyDown(Enum.KeyCode.Space) then m+=Vector3.new(0,1,0) end
            if u:IsKeyDown(Enum.KeyCode.LeftShift) then m-=Vector3.new(0,1,0) end
            bv.Velocity=m*speed end bg:Destroy() bv:Destroy() h.PlatformStand=false end) end end})
    
    PTab:CreateToggle({Name="🛡 God Mode",CurrentValue=false,Flag="gd",
        Callback=function(v) godModeValue=v if v then task.spawn(function() while godModeValue do
            task.wait(prem and 0.05 or 0.1) local c=plr.Character
            if c then local h=c:FindFirstChild("Humanoid") if h and h.Health<h.MaxHealth then h.Health=h.MaxHealth end end end end) end end})
    
    if prem then PTab:CreateToggle({Name="👻 Noclip",CurrentValue=false,Flag="nc",
        Callback=function(v) task.spawn(function() while v do task.wait(0.1)
            local c=plr.Character if c then for _,p in c:GetDescendants() do if p:IsA("BasePart") then p.CanCollide=false end end end end end) end}) end
    
    local ETab = Window:CreateTab("👁 ESP", "eye")
    local esp={}
    ETab:CreateToggle({Name="👁 ESP "..(prem and"[CHAMS]"or""),CurrentValue=false,Flag="esp",
        Callback=function(v)
            if v then for _,p2 in game.Players:GetPlayers() do if p2~=plr and p2.Character then
                local hl=Instance.new("Highlight",p2.Character)
                hl.FillColor=prem and Color3.fromRGB(0,255,255) or Color3.fromRGB(255,0,0)
                hl.OutlineColor=Color3.fromRGB(255,255,255) hl.FillTransparency=prem and 0.3 or 0.5
                table.insert(esp,{p=p2,h=hl}) end end
            game.Players.PlayerAdded:Connect(function(p2) p2.CharacterAdded:Connect(function(c) task.wait(0.5)
                local hl=Instance.new("Highlight",c) hl.FillColor=prem and Color3.fromRGB(0,255,255) or Color3.fromRGB(255,0,0)
                hl.OutlineColor=Color3.fromRGB(255,255,255) hl.FillTransparency=prem and 0.3 or 0.5
                table.insert(esp,{p=p2,h=hl}) end) end)
            game.Players.PlayerRemoving:Connect(function(p2) for i,o in esp do if o.p==p2 then o.h:Destroy() table.remove(esp,i) break end end end)
            else for _,o in esp do o.h:Destroy() end esp={} end end})
    
    local MTab = Window:CreateTab("🔧 Khác", "settings")
    MTab:CreateButton({Name="🔧 Infinite Yield",Callback=function() loadstring(game:HttpGet("https://raw.githubusercontent.com/EdgeIY/infiniteyield/master/source"))() end})
    MTab:CreateButton({Name="🔄 Rejoin",Callback=function() game:GetService("TeleportService"):Teleport(game.PlaceId,plr) end})
    MTab:CreateButton({Name="🌐 Server Hop",Callback=function()
        local s=game:GetService("HttpService"):JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/"..game.PlaceId.."/servers/Public?sortOrder=Asc&limit=100"))
        if s and s.data then for _,v in s.data do if v.playing<v.maxPlayers and v.id~=game.JobId then
            game:GetService("TeleportService"):TeleportToPlaceInstance(game.PlaceId,v.id) break end end end
    end})
    
    Rayfield:Notify({
        Title = prem and "👑 PRIME" or "⚔️ VIP",
        Content = "Auto Move Kill All đã sẵn sàng!",
        Duration = 5, Image = "check"
    })
    print("✅ DEV_MINH 2005 - Auto Move Kill All Loaded!")
end
